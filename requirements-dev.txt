# packaged into a Lambda -- src/requirements.txt is what ships.

# boto3/botocore are deliberately NOT in src/requirements.txt: the Lambda
# runtime provides them, and shipping a second copy in the bundle is wasted

# here `pytest` cannot even collect, which is how the first CI run of the test
# step failed -- in one second, before a single test executed.

# Marengo managed body (modelConfiguration and supplemental storage).
# Tests load that model. The Lambda runtime's copy is older; the functions
# that call the API install src/requirements-bedrock-kb.txt instead.
boto3>=1.43.92
botocore>=1.43.92

PyYAML>=6.0

pytest>=8.0

# tests/test_agentcore_infra.py checks the template against. Without it that

cfn-lint
