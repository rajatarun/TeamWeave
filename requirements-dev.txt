# Test-only dependencies, kept out of requirements.txt so none of this is
# packaged into a Lambda -- src/requirements.txt is what ships.
#
# boto3/botocore are deliberately NOT in src/requirements.txt: the Lambda
# runtime provides them, and shipping a second copy in the bundle is wasted
# space. But five test modules import them at module scope, so without them
# here `pytest` cannot even collect, which is how the first CI run of the test
# step failed -- in one second, before a single test executed.
# 1.43.92 is the first botocore whose bedrock-agent model accepts the
# managed body this stack sends (supplemental storage and the managed
# data-source connector).
# Tests load that model. The Lambda runtime's copy is older; the functions
# that call the API install src/requirements-bedrock-kb.txt instead.
boto3>=1.43.92
botocore>=1.43.92

# tests/test_openapi_contract.py parses openapi/teamweave.yaml.
PyYAML>=6.0

pytest>=8.0

# tests/test_template_lint.py runs the linter on infra/template.yaml so an
# error such as E3034 fails pytest rather than the later deploy-time
# `sam validate --lint`. Without the package the AgentCore test skips, and
# a skipped test in CI is not a test.
cfn-lint
